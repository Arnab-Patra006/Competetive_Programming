#### 6. [Generate all binary strings](https://www.geeksforgeeks.org/problems/generate-all-binary-strings/1)
```
class Solution {
  public:
  void solve(int i,int n,vector<string>&ans,string curr){
      if(i>n){
          ans.push_back(curr);
          return;
      }
      string curr0=curr+"0";
      solve(i+1,n,ans,curr0); //if put 0 first no need to sort anymore
      string curr1=curr+"1";
      solve(i+1,n,ans,curr1);
      return;
  }
    vector<string> binstr(int n) {
        vector<string>ans;
        solve(1,n,ans,"");
        return ans;
    }
};
```
